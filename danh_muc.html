Xử lý đọc file Excel và đẩy lên Supabase (Đã thêm bộ lọc chống trùng lặp) async function docFileExcel(event) { const file = event.target.files[0]; if (!file) return; const reader = new FileReader(); reader.onload = async function(e) { try { const data = new Uint8Array(e.target.result); const workbook = XLSX.read(data, {type: 'array'}); // Lấy sheet đầu tiên const firstSheet = workbook.Sheets[workbook.SheetNames[0]]; const jsonData = XLSX.utils.sheet_to_json(firstSheet, {defval: null}); if (jsonData.length === 0) { alert("File Excel trống, không có dữ liệu!"); return; } const tenBang = document.getElementById('chonBang').value; // TỰ ĐỘNG LỌC TRÙNG LẶP DỮ LIỆU let khoaChinh = ""; if (tenBang === 'khach_hang') khoaChinh = 'ma_kh'; else if (tenBang === 'hang_hoa' || tenBang === 'dinh_khoan_hang_hoa') khoaChinh = 'ma_hang'; else if (tenBang === 'danh_sach_xe') khoaChinh = 'ma_xe'; let duLieuSach = []; if (khoaChinh) { const map = new Map(); jsonData.forEach(item => { // Loại bỏ khoảng trắng thừa ở mã và giữ lại dòng cuối cùng nếu trùng if (item[khoaChinh]) { item[khoaChinh] = String(item[khoaChinh]).trim(); map.set(item[khoaChinh], item); } }); duLieuSach = Array.from(map.values()); } else { duLieuSach = jsonData; } if(!confirm(`Tìm thấy ${jsonData.length} dòng (Sau khi lọc trùng còn ${duLieuSach.length} dòng).\nBạn có chắc muốn chèn vào bảng "${tenBang}"?`)) { event.target.value = ''; return; } // Đẩy dữ liệu sạch lên Supabase const { error } = await sb.from(tenBang).upsert(duLieuSach); if (error) throw error; alert("Tải dữ liệu lên thành công!"); taiDuLieu(); } catch (error) { alert("Lỗi khi tải file: " + error.message + "\n\nLƯU Ý: Tên cột (dòng 1) trong Excel phải viết thường và khớp chữ 100% với tên cột trên màn hình web."); } finally { event.target.value = ''; } }; reader.readAsArrayBuffer(file); }
